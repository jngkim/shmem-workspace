cmake_minimum_required(VERSION 3.20)

project(ishmem_omp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_OPENSHMEM "Enable OpenSHMEM RT" FALSE)
if(ENABLE_OPENSHMEM)
  if(DEFINED SHMEM_DIR)
    set(ENV{PKG_CONFIG_PATH} "${SHMEM_DIR}/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
  endif()
  find_package(PkgConfig REQUIRED)
  add_library(sos INTERFACE)
  pkg_check_modules(SANDIA_SHMEM REQUIRED IMPORTED_TARGET sandia-openshmem)
  string(REGEX REPLACE "-Wl,[^;]*;" "" SANDIA_SHMEM_LDFLAGS "${SANDIA_SHMEM_LDFLAGS}")
  target_include_directories(sos INTERFACE ${SANDIA_SHMEM_INCLUDE_DIRS})
  target_link_libraries(sos INTERFACE ${SANDIA_SHMEM_LDFLAGS})
endif()

if(DEFINED ISHMEM_DIR AND EXISTS "${ISHMEM_DIR}/lib/cmake/ishmem/ISHMEMConfig.cmake")
  set(ISHMEM_DIR "${ISHMEM_DIR}/lib/cmake/ishmem" CACHE PATH "ISHMEM config directory" FORCE)
endif()
find_package(ISHMEM CONFIG REQUIRED)

add_library(ishmem INTERFACE)
target_link_libraries(ishmem INTERFACE ISHMEM::HOST_SHARED)
if(TARGET sos)
  target_link_libraries(ishmem INTERFACE sos)
endif()

function(add_sycl_app target_name)
  add_executable(${target_name} ${ARGN})
  target_compile_options(${target_name} PRIVATE -fsycl)
  target_link_options(${target_name} PRIVATE -fsycl)
  target_link_libraries(${target_name} PRIVATE ishmem)
endfunction()

function(add_omp_app target_name)
  add_executable(${target_name} ${ARGN})
  target_compile_options(${target_name} PRIVATE -fiopenmp -fopenmp-targets=spir64 -fsycl)
  target_link_options(${target_name} PRIVATE -fiopenmp -fopenmp-targets=spir64)
  target_link_libraries(${target_name} PRIVATE ishmem)
endfunction()

add_sycl_app(test_init test_init.cpp)
add_sycl_app(test_sycl test_sycl.cpp)
add_omp_app(test_omp test_omp.cpp)
